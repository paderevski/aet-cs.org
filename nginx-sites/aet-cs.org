server {
	listen [::]:443 ssl;
	listen 443 ssl;
	server_name aet-cs.org;

	root /home/bitnami/aet-cs.org/dist;
	index index.html index.htm index.nginx-debian.html;

	location / {
		try_files $uri $uri/ index.html;
	}

	ssl_certificate /etc/letsencrypt/live/aet-cs.org/fullchain.pem;
	ssl_certificate_key /etc/letsencrypt/live/aet-cs.org/privkey.pem;
	include /etc/letsencrypt/options-ssl-nginx.conf;
	ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}

	listen 80;
	listen [::]:80;
	server_name aet-cs.org;
	return 301 https://$host$request_uri;
